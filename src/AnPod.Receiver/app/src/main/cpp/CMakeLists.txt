cmake_minimum_required(VERSION 3.22.1)
project(anpod_oboe_output LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED YES)
set(CMAKE_CXX_EXTENSIONS NO)

if(ANDROID)
    find_package(oboe REQUIRED CONFIG)
    add_library(anpod_oboe_output SHARED OboeNativeBridge.cpp)
    target_link_libraries(anpod_oboe_output PRIVATE oboe::oboe log)
    target_link_options(anpod_oboe_output PRIVATE "-Wl,-z,max-page-size=16384")
else()
    enable_testing()
    find_package(Threads REQUIRED)
    add_executable(spsc_float_ring_test SpscFloatRingTest.cpp)
    target_compile_definitions(spsc_float_ring_test PRIVATE ANPOD_RING_TESTING)
    if(MSVC)
        target_compile_options(spsc_float_ring_test PRIVATE /UNDEBUG)
    else()
        target_compile_options(spsc_float_ring_test PRIVATE -UNDEBUG)
    endif()
    target_link_libraries(spsc_float_ring_test PRIVATE Threads::Threads)
    add_test(NAME spsc_float_ring_test COMMAND spsc_float_ring_test)
    set_tests_properties(spsc_float_ring_test PROPERTIES TIMEOUT 10)
endif()
